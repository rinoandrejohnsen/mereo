# calls.mereo -- primitive declarations for the Linux x86-64 syscall ABI.
# The mereo equivalent of linux/calls.h++: names, numbers, and port shapes.
# Numbers verified against linux/calls.h++, the project's canonical table.
#
# A port is declared by its direction: `PORT in` = a value argument,
# `PORT out` = the caught return, `@PORT` = an address argument (a pointer
# to a buffer is passed). `final call` marks noreturn.
#
# Import this file with a use-line naming calls.mereo.

asm.mereo

# asm.mereo -- raw inline-asm primitives, the general form of `call`.
# A `call N` is sugar for an `asm "syscall"`; here the template and the
# per-operand register constraints are written out. Directions are the
# same verbs as everywhere: is = input, into = output.

use "calls.mereo"

# Pure computation: no side effects, so `pure` (non-volatile) lets GCC
# fold/hoist/eliminate it. popcnt reads one register, writes another.
population_count is pure asm "popcnt %[source], %[result]" where
  source in register
  result out register

# A full memory barrier: no operands, only the memory clobber. Volatile
# by default (a barrier that GCC could delete would be worthless).
memory_fence is asm "mfence" where
  clobbers memory

# Hardware RNG into a register; the carry flag (cc) says whether the
# value is valid, so it is clobbered.
random_word is asm "rdrand %[result]" where
  result out register
  clobbers cc

linux.mereo

# linux.mereo -- conceptual equivalents of linux.h++, in mereo.
# Imports its own primitive declarations; programs need only
# `use "linux.mereo"`.
#
# What ports directly (linux.h++ -> here):
#   file_descriptor         ->  file_descriptor   THE ACQUIRED RESOURCE:
#                                        its lifecycle is open (initialize)
#                                        and close (finalize), nothing else.
#                                        Ownable directly, adoptable, and
#                                        composable.
#   file                    ->  file     what you DO with the resource: it
#                                        COMPOSES a file_descriptor part
#                                        (`handle is file_descriptor`) and
#                                        carries the I/O methods, reaching
#                                        the fd through the part path
#                                        `handle descriptor`. Its initialize
#                                        opens (a constructor chain to the
#                                        part); no finalize -- the part's
#                                        close is propagated. Construct one
#                                        to open (`f is file where path ...`)
#                                        or adopt a raw descriptor
#                                        (`f is already file where
#                                        descriptor is 1` -- the leaf name
#                                        addresses the composed slot).
#   mapping                 ->  mapping  with a class-constant length: a
#                                        mereo constructor cannot yet
#                                        persist a parameter into state,
#                                        and munmap needs the length.
#                                        Declare a class per size, or
#                                        wait for param-to-state.
#   process (kill, running) ->  process  adopted-only: wrap a known pid.
#
# Skipped, with reasons (each is a known design thread):
#   duplicate, accept, pidfd   factory methods returning a new owned
#                              instance (instance-valued ports)
#   release, detach            ownership transfer; mereo ownership is
#                              fixed at declaration
#   subprocess spawn           fork returns twice; needs the fused spawn
#                              pseudo-primitive or branches
#   socket                     the class would port, but using it needs
#                              sockaddr bytes the language cannot
#                              construct yet
#   app                        argv and env live in initial-stack pointer
#                              land; its status-then-exit-last design IS
#                              mereo already (the implicit status and the
#                              release tower)
#
# Contract caveat: mmap's true failure test is the [-4095,-1] window;
# `address > 0` is correct for user mappings on x86-64 until an
# `ensure ok` form exists.

use "calls.mereo"

file_descriptor is
  descriptor is -1

  initialize with path and flags and mode is
    open where
      path is path
      flags is flags
      mode is mode
      descriptor is descriptor
    ensure descriptor >= 0

  finalize is
    close where
      descriptor is descriptor

file is
  handle is file_descriptor

  initialize with path and flags and mode is
    initialize handle where
      path is path
      flags is flags
      mode is mode

  read with buffer and capacity and count is
    read where
      descriptor is handle descriptor
      buffer is @buffer
      capacity is capacity
      count is count
    ensure count >= 0

  write with buffer and count and written is
    write where
      descriptor is handle descriptor
      buffer is @buffer
      count is count
      written is written
    ensure
      written >= 0
      written == count

  seek with offset and whence and position is
    lseek where
      descriptor is handle descriptor
      offset is offset
      whence is whence
      position is position
    ensure position >= 0

  sync with result is
    fsync where
      descriptor is handle descriptor
      result is result
    ensure result >= 0

mapping is
  address is 0
  length is 4096

  initialize with protection is
    mmap where
      hint is 0
      length is length
      protection is protection
      flags is 34
      descriptor is -1
      offset is 0
      address is address
    ensure address > 0

  finalize is
    munmap where
      address is address
      length is length

process is
  pid is -1

  kill with signal and result is
    kill where
      pid is pid
      signal is signal
      result is result
    ensure result >= 0

  probe with result is
    kill where
      pid is pid
      signal is 0
      result is result
    ensure result >= 0

socket.mereo

use "linux.mereo"

# Socket demo: connect a TCP socket to a local server, read the bytes it
# sends, and write them to the terminal. `connection` is a resource whose
# lifecycle is socket() to open and close() to release; `reach` runs
# connect(). Everything is a file descriptor, so once connected we read()
# it like any file.
#
# The one hard part is the peer address. mereo cannot build a struct, but
# a string literal IS a byte array in C, so the 16-byte sockaddr_in is
# spelled out as raw bytes below:
#   \x02\x00            AF_INET (family 2, little-endian)
#   \x30\x39            port 12345 (0x3039, network/big-endian)
#   \x7f\x00\x00\x01    address 127.0.0.1
#   \x00 * 8            padding

connection is
  descriptor is -1
  status is 0

  initialize with domain and type and protocol is
    socket where
      domain is domain
      type is type
      protocol is protocol
      descriptor is descriptor
    ensure descriptor >= 0

  reach with address and length is
    connect where
      descriptor is descriptor
      address is address
      length is length
      result is status
    ensure status >= 0

  read with buffer and capacity and count is
    read where
      descriptor is descriptor
      buffer is @buffer
      capacity is capacity
      count is count
    ensure count >= 0

  finalize is
    close where
      descriptor is descriptor

program is
  capacity is 4096
  buffer is capacity bytes
  count is 0
  written is 0

  terminal is already file where
    descriptor is 1

  link is connection where
    domain is 2
    type is 1
    protocol is 0

  reach link where
    address is "\x02\x00\x30\x39\x7f\x00\x00\x01\x00\x00\x00\x00\x00\x00\x00\x00"
    length is 16

  read link where
    buffer is @buffer
    capacity is capacity
    count is count

  write terminal where
    buffer is @buffer
    count is count
    written is written

  exit

server.mereo

use "linux.mereo"

# Server demo: bind a TCP socket, listen, then loop, accepting
# connections, reading what each client sends, and writing it to the
# terminal. `accept` hands back a new fd; `client is adopted connection`
# takes ownership, and because it is acquired inside the `loop is` body it
# is closed at the end of every iteration -- the loop is liveness-neutral.
# The listening socket stays live across iterations; the tower closes it
# (and the current client) only if a syscall faults and the server exits.
#
# bind address = sockaddr_in: AF_INET, port 12346, 127.0.0.1
#   \x02\x00  family      \x30\x3a  port 12346      \x7f\x00\x00\x01  addr

connection is
  descriptor is -1
  status is 0

  initialize with domain and type and protocol is
    socket where
      domain is domain
      type is type
      protocol is protocol
      descriptor is descriptor
    ensure descriptor >= 0

  bind with address and length is
    bind where
      descriptor is descriptor
      address is address
      length is length
      result is status
    ensure status >= 0

  listen with backlog is
    listen where
      descriptor is descriptor
      backlog is backlog
      result is status
    ensure status >= 0

  accept with accepted is
    accept where
      descriptor is descriptor
      address is 0
      length is 0
      connection is accepted
    ensure accepted >= 0

  read with buffer and capacity and count is
    read where
      descriptor is descriptor
      buffer is @buffer
      capacity is capacity
      count is count
    ensure count >= 0

  finalize is
    close where
      descriptor is descriptor

program is
  capacity is 4096
  buffer is capacity bytes
  count is 0
  written is 0
  peer is -1

  terminal is already file where
    descriptor is 1

  server is connection where
    domain is 2
    type is 1
    protocol is 0

  bind server where
    address is "\x02\x00\x30\x3a\x7f\x00\x00\x01\x00\x00\x00\x00\x00\x00\x00\x00"
    length is 16

  listen server where
    backlog is 1

  loop is
    accept server where
      accepted is peer

    client is adopted connection where
      descriptor is peer

    read client where
      buffer is @buffer
      capacity is capacity
      count is count

    write terminal where
      buffer is @buffer
      count is count
      written is written

  exit

visit.mereo

use "calls.mereo"

# visit demo: read stdin, then branch on how many bytes arrived. Unlike the
# old switch (every case cold), a `visit` promotes its DEFAULT arm onto the C
# hot path -- `many` falls straight through, no jump -- while the guarded arms
# are cold detours that run and return to the same place. Every arm body is
# defined below `exit`; the compiler decides the layout.
#
#   printf ''    | ./visit_bin   -> nothing
#   printf 'x'   | ./visit_bin   -> one byte
#   printf 'xyz' | ./visit_bin   -> many bytes   (the hot-path default)

file_descriptor is
  descriptor is -1

  read with buffer and capacity and count is
    read where
      descriptor is descriptor
      buffer is @buffer
      capacity is capacity
      count is count
    ensure count >= 0

  write with buffer and count and written is
    write where
      descriptor is descriptor
      buffer is @buffer
      count is count
      written is written
    ensure written >= 0

program is
  capacity is 4096
  buffer is capacity bytes
  count is 0
  written is 0

  input is already file_descriptor where
    descriptor is 0
  terminal is already file_descriptor where
    descriptor is 1

  read input where
    buffer is @buffer
    capacity is capacity
    count is count

  visit many or nothing if count is 0 or single if count is 1

  exit

  many is
    write terminal where
      buffer is "many bytes\n"
      count is 11
      written is written
  nothing is
    write terminal where
      buffer is "nothing\n"
      count is 8
      written is written
  single is
    write terminal where
      buffer is "one byte\n"
      count is 9
      written is written

span.mereo

use "calls.mereo"

# span -- mereo's string_view: a transparent, non-owning view holding just a
# pointer (`data`) and a `length`. It owns nothing; it just names a region.
# Read stdin, view the bytes that arrived with a span, and write the span
# out -- the write takes ONE thing (a view) instead of a buffer and a count.
#
#   echo hello | ./span_bin   ->  hello
#
# `data is @buffer` is the adoption tweak: a state slot set to a buffer's
# ADDRESS. `already` names the view from given values; being a pure
# value view (no resource), there is nothing for the tower to release.

span is
  data is 0
  length is 0

  show with descriptor and written is
    write where
      descriptor is descriptor
      buffer is data # a scalar holding the pointer -> passed raw
      count is length
      written is written
    ensure written >= 0

file_descriptor is

  read with buffer and capacity and count is
    read where
      descriptor is descriptor
      buffer is @buffer
      capacity is capacity
      count is count
    ensure count >= 0

program is

  count is 0
  written is 0

    descriptor is 0

  read input where
    buffer is @buffer
    capacity is capacity
    count is count

  view is already span where
    data is @buffer
    length is count

  show view where
    descriptor is 1
